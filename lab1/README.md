## Практические задания

### Пример 1. Композиция: классы «Дом» и «Комната»

```cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

class Room {
private:
    string name;
    double area;

public:
    Room(const string& n, double a) : name(n), area(a) {
        cout << "Создана комната \"" << name << "\" площадью " << area << " м²" << endl;
    }

    ~Room() {
        cout << "Удалена комната \"" << name << "\"" << endl;
    }

    void print() const {
        cout << " - " << name << ": " << area << " м²" << endl;
    }
};

class House {
private:
    string address;
    vector<Room> rooms; // композиция: комнаты внутри дома

public:
    House(const string& addr) : address(addr) {
        cout << "Построен дом по адресу: " << address << endl;
    }

    ~House() {
        cout << "Снесён дом по адресу: " << address << endl;
    }

    void addRoom(const string& name, double area) {
        rooms.emplace_back(name, area); // комната создаётся внутри дома
    }

    void print() const {
        cout << "Дом по адресу " << address << ":" << endl;
        for (const auto& room : rooms) {
            room.print();
        }
    }
};

int main() {
    setlocale(LC_ALL, "ru");
    House myHouse("ул. Цветочная, д. 15");

    myHouse.addRoom("Гостиная", 25.5);
    myHouse.addRoom("Кухня", 12.0);
    myHouse.addRoom("Спальня", 18.0);

    myHouse.print();

    return 0;
}
```

```

Создана комната "Спальня" площадью 18 м²
Дом по адресу ул. Цветочная, д. 15:
 - Гостиная: 25.5 м²
 - Кухня: 12 м²

Снесён дом по адресу: ул. Цветочная, д. 15

Удалена комната "Гостиная"
```

Пример 2. Агрегация: классы «Библиотека» и «Книга»
Задача: Реализовать классы Book и Library. Книги могут существовать без библиотеки (агрегация).

```cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

class Book {
private:
    string title;
    string author;

public:
    Book(const string& t, const string& a) : title(t), author(a) {
        cout << "Создана книга \"" << title << "\" (" << author << ")" << endl;
    }

    ~Book() {
        cout << "Удалена книга \"" << title << "\"" << endl;
    }

    string getTitle() const { return title; }

    void print() const {
        cout << " - \"" << title << "\", " << author << endl;
    }
};

class Library {
private:
    string name;
    vector<Book*> books;

public:
    Library(const string& n) : name(n) {
        cout << "Открыта библиотека \"" << name << "\"" << endl;
    }

    ~Library() {
        cout << "Закрыта библиотека \"" << name << "\"" << endl;
    }

    void addBook(Book* book) {
        books.push_back(book);
        cout << "Книга \"" << book->getTitle() << "\" добавлена в библиотеку" << endl;
    }

    void print() const {
        cout << "Библиотека \"" << name << "\" содержит:" << endl;
        for (const auto* book : books) {
            book->print();
        }
    }
};

int main() {
    setlocale(LC_ALL, "ru");

    Book b1("Война и мир", "Л. Толстой");
    Book b2("Преступление и наказание", "Ф. Достоевский");
    Book b3("Евгений Онегин", "А. Пушкин");

    {
        Library lib("Городская библиотека");
        lib.addBook(&b1);
        lib.addBook(&b2);
        lib.print();
    } 

    cout << "\nКниги всё ещё существуют после закрытия библиотеки:" << endl;
    b1.print();
    b2.print();

    return 0;
}
```

Пример 3. Статические члены: счётчик объектов
Задача: Реализовать класс Employee со статическим полем для подсчёта количества созданных объектов.

```cpp
#include <iostream>
#include <string>
using namespace std;

class Employee {
private:
    string name;
    double salary;
    static int employeeCount;
    static double totalSalary;

public:
    Employee(const string& n, double s) : name(n), salary(s) {
        employeeCount++;
        totalSalary += salary;
        cout << "Создан сотрудник: " << name << ", зарплата: " << salary << endl;
    }

    ~Employee() {
        employeeCount--;
        totalSalary -= salary;
        cout << "Уволен сотрудник: " << name << endl;
    }

    static int getCount() {
        return employeeCount;
    }

    static double getAverageSalary() {
        if (employeeCount == 0) return 0;
        return totalSalary / employeeCount;
    }

    void setSalary(double newSalary) {
        totalSalary = totalSalary - salary + newSalary;
        salary = newSalary;
    }

    void print() const {
        cout << " - " << name << ": " << salary << " руб." << endl;
    }
};

int Employee::employeeCount = 0;
double Employee::totalSalary = 0;

int main() {
    setlocale(LC_ALL, "ru");

    cout << "Всего сотрудников: " << Employee::getCount() << endl;
    cout << "Средняя зарплата: " << Employee::getAverageSalary() << endl;

    Employee e1("Иванов Иван", 50000);
    Employee e2("Петров Пётр", 60000);
    Employee e3("Сидорова Мария", 55000);

    cout << "\nВсего сотрудников: " << Employee::getCount() << endl;
    cout << "Средняя зарплата: " << Employee::getAverageSalary() << endl;

    cout << "\nСписок сотрудников:" << endl;
    e1.print();
    e2.print();
    e3.print();

    e2.setSalary(65000);
    cout << "\nПосле повышения зарплаты Петрова:" << endl;
    cout << "Средняя зарплата: " << Employee::getAverageSalary() << endl;

    return 0;
}
```

```cpp
        : name(n), position(pos), salary(sal) {
        count++;
        cout << "Создан сотрудник \"" << name << "\"" << endl;
    }

    ~Employee() {
        count--;
        cout << "Удалён сотрудник \"" << name << "\"" << endl;

    void print() const {
        cout << " - " << name << ", должность: " << position

    void print() const {
        cout << "Отдел \"" << name << "\" содержит:" << endl;
```

**Контрольные вопросы:**

Композиция — это строгая связь «часть-целое», при которой часть не может существовать отдельно от целого. Пример: дом и его комнаты, автомобиль и его двигатель.

Агрегация — это более слабая связь, при которой часть может существовать независимо от целого. Отличие в том, что при композиции целое управляет жизнью частей, а при агрегации — нет.

Композиция в C++ реализуется хранением объектов по значению (как обычные поля-объекты) внутри класса.

Агрегация в C++ реализуется хранением указателей или ссылок на объекты, которые создаются и удаляются где-то снаружи.

Разница во времени жизни: при композиции части создаются и уничтожаются вместе с целым; при агрегации части живут независимо и могут существовать до и после целого.

Статическое поле класса — это поле, принадлежащее самому классу, а не объекту. Оно существует в одном экземпляре для всех объектов, в отличие от обычного поля, которое у каждого объекта своё.

Статическое поле объявляется внутри класса с ключевым словом static, а определяется (выделяется память) вне класса. Оно хранится в сегменте данных программы, а не внутри объектов.

Статический метод — это метод, принадлежащий классу, а не объекту; его можно вызывать без создания объекта. Из него нельзя обращаться к нестатическим полям, потому что он не привязан к конкретному объекту и не имеет указателя this.

Статические счётчики объектов используются для подсчёта количества существующих в данный момент объектов класса: увеличиваются в конструкторе и уменьшаются в деструкторе.

Паттерн «Одиночка» (Singleton) — это паттерн, гарантирующий существование только одного экземпляра класса и предоставляющий глобальную точку доступа к нему. Нужен для управления общими ресурсами (настройки, логгер, соединение с БД).

Запретить копирование объекта можно, объявив конструктор копирования и оператор присваивания как = delete (в современном C++).

Порядок вызовов при композиции: конструкторы полей-объектов вызываются в порядке их объявления в классе (до тела конструктора), а деструкторы — в строго обратном порядке.

Да, из обычного метода можно изменить статическое поле, так как нестатический метод имеет доступ ко всем полям класса.

Статические поля нужно определять вне класса, потому что объявление внутри класса не выделяет память; реальное определение требуется для выделения памяти и избежания ошибки линковки.

Композицию стоит использовать, когда часть не имеет смысла без целого (сердце и человек). Агрегацию — когда объекты связаны логически, но независимы, или один объект может принадлежать нескольким целым (студент и университет).
