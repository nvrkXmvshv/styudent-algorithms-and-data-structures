```cpp
#include <iostream>
#include <utility> 
#include <stdexcept>

using namespace std;

class Vector {
private:
    int* data;      
    size_t size;   
    size_t capacity; 

    void reallocate(size_t new_capacity) {
        int* new_data = new int[new_capacity];
        for (size_t i = 0; i < size; ++i) {
            new_data[i] = data[i];
        }
        delete[] data;
        data = new_data;
        capacity = new_capacity;
    }

public:
    Vector() : data(nullptr), size(0), capacity(0) {}

    Vector(size_t initial_size, int value = 0) : size(initial_size), capacity(initial_size) {
        data = new int[capacity];
        for (size_t i = 0; i < size; ++i) {
            data[i] = value;
        }
    }

    ~Vector() {
        delete[] data;
        cout << "Деструктор: память освобождена." << endl;
    }

    Vector(const Vector& other) : size(other.size), capacity(other.capacity) {
        data = new int[capacity];
        for (size_t i = 0; i < size; ++i) {
            data[i] = other.data[i];
        }
        cout << "Конструктор копирования вызван." << endl;
    }

    Vector(Vector&& other) noexcept 
        : data(other.data), size(other.size), capacity(other.capacity) {
        other.data = nullptr;
        other.size = 0;
        other.capacity = 0;
        cout << "Конструктор перемещения вызван." << endl;
    }

    Vector& operator=(const Vector& other) {
        if (this != &other) { 
            Vector temp(other); 
            swap(temp);         
        } 
        cout << "Оператор копирующего присваивания вызван." << endl;
        return *this;
    }

    Vector& operator=(Vector&& other) noexcept {
        if (this != &other) {
            delete[] data; 
            
            data = other.data;
            size = other.size;
            capacity = other.capacity;

            other.data = nullptr;
            other.size = 0;
            other.capacity = 0;
        }
        cout << "Оператор перемещающего присваивания вызван." << endl;
        return *this;
    }

    void swap(Vector& other) noexcept {
        std::swap(data, other.data);
        std::swap(size, other.size);
        std::swap(capacity, other.capacity);
    }

    void push_back(int value) {
        if (size >= capacity) {
            size_t new_cap = (capacity == 0) ? 1 : capacity * 2;
            reallocate(new_cap);
        }
        data[size++] = value;
    }

    // Индивидуальное задание: метод concat
    Vector concat(const Vector& other) const {
        Vector result;
        result.capacity = this->size + other.size;
        result.data = new int[result.capacity];
        
        for (size_t i = 0; i < this->size; ++i) {
            result.data[i] = this->data[i];
        }

        for (size_t i = 0; i < other.size; ++i) {
            result.data[this->size + i] = other.data[i];
        }
        result.size = this->size + other.size;
        
        return result;
    }

    void print() const {
        cout << "[ ";
        for (size_t i = 0; i < size; ++i) {
            cout << data[i] << " ";
        }
        cout << "]" << endl;
    }
};
```

Контрольные вопросы

Динамический массив называется так, потому что его размер (и выделенная память) не фиксируется на этапе компиляции, а определяется во время выполнения программы. Он может расти или уменьшаться в процессе работы.

Размер (size) — это количество элементов, которые фактически хранятся в массиве в данный момент. Ёмкость (capacity) — это общее количество элементов, под которые уже выделена память. Размер всегда меньше или равен ёмкости.

При удвоении ёмкости средняя стоимость добавления одного элемента остаётся постоянной. Если увеличивать ёмкость на фиксированное число, то при каждом добавлении придётся копировать все элементы заново, что приведёт к квадратичной сложности в долгосрочной перспективе.

Амортизированная сложность операции push_back — O(1). Хотя иногда операция занимает O(n) (при расширении и копировании), эти дорогие операции происходят достаточно редко, и в среднем на каждое добавление тратится константное время.

Правило трёх гласит, что если классу нужен хотя бы один из трёх методов (деструктор, конструктор копирования или оператор копирующего присваивания), то нужны все три. Применяется, когда класс управляет ресурсами вручную (например, сырыми указателями).

Правило пяти добавляет к правилу трёх конструктор перемещения и оператор перемещающего присваивания. Оно применяется в современном C++ (начиная с C++11) для оптимизации работы с временными объектами и избежания лишнего копирования.

Конструктор копирования создаёт новый объект на основе существующего (вызывается при инициализации). Оператор присваивания копирует данные в уже существующий объект, у которого ранее была выделена память (которую перед этим нужно освободить).

Проверка if (this != &other) защищает от самоприсваивания (когда объект присваивается сам себе). Без неё объект может сначала освободить свою память, а затем попытаться скопировать данные из самого себя, что приведёт к чтению из освобождённой памяти и ошибке.

Итератор — это объект, который предоставляет доступ к элементам контейнера и позволяет перемещаться по ним. По сути своей работы итератор очень похож на указатель: к нему применимы операции разыменования и инкремента, но он может быть умнее (например, проверять границы).

Методы begin() и end() задают диапазон для перебора элементов контейнера. begin() возвращает итератор на первый элемент, а end() — на позицию, следующую за последним элементом. Это стандартный способ работы с алгоритмами STL и циклами.

Для реализации range-based for (цикла for (auto x : obj)) в классе нужно реализовать методы begin() и end(), возвращающие итераторы (или указатели), а также операторы инкремента, разыменования и сравнения для этих итераторов.

Глубокое копирование создаёт полностью независимую копию объекта, включая копирование всех данных по указателям (выделяется новая память). Поверхностное копирование просто копирует значения указателей, из-за чего два объекта начинают указывать на одну и ту же область памяти.

Семантика перемещения позволяет «забрать» ресурсы у временного или ненужного объекта (например, переместить указатель на массив), вместо того чтобы копировать данные. Это нужно для значительного ускорения работы программы и экономии памяти.

Конструктор перемещения помечается noexcept, чтобы гарантировать, что он не выбросит исключение. Это важно для стандартных контейнеров (например, std::vector), так как они принимают решение о перемещении или копировании элементов при изменении размера на основе этого флага.

В деструкторе память, выделенная через new[], должна освобождаться через delete[]. Важно проверить указатель на nullptr (освобождение nullptr безопасно) и обнулить его после удаления, чтобы избежать повторного удаления.
