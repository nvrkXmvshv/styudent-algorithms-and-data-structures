#include <iostream>
#include <stack>
#include <queue>
#include <deque>
#include <vector>

using namespace std;

// ==========================================
// 1. СТЕК: Удалить все нечётные элементы
// ==========================================
void taskStack() {
    cout << "--- 1. СТЕК (удаление нечетных) ---" << endl;
    stack<int> originalStack;
    
    // Заполняем стек (снизу вверх: 1, 2, 3, 4, 5)
    for (int i = 1; i <= 5; ++i) originalStack.push(i);
    
    stack<int> tempStack;
    // Извлекаем элементы и оставляем только чётные
    while (!originalStack.empty()) {
        int val = originalStack.top();
        originalStack.pop();
        if (val % 2 == 0) {
            tempStack.push(val);
        }
    }
    
    // Возвращаем элементы обратно в исходный стек, чтобы сохранить порядок
    while (!tempStack.empty()) {
        originalStack.push(tempStack.top());
        tempStack.pop();
    }
    
    // Вывод результата
    cout << "Стек после удаления нечетных: ";
    while (!originalStack.empty()) {
        cout << originalStack.top() << " ";
        originalStack.pop();
    }
    cout << "\n\n";
}

// ==========================================
// 2. ОЧЕРЕДЬ: Проверить на палиндром
// ==========================================
void taskQueue() {
    cout << "--- 2. ОЧЕРЕДЬ (проверка на палиндром) ---" << endl;
    queue<int> q1, q2;
    
    // Тест 1: Палиндром (1, 2, 3, 2, 1)
    int arr1[] = {1, 2, 3, 2, 1};
    for (int x : arr1) q1.push(x);
    
    // Тест 2: Не палиндром (1, 2, 3, 4, 5)
    int arr2[] = {1, 2, 3, 4, 5};
    for (int x : arr2) q2.push(x);
    
    auto isPalindrome = [](queue<int> q) {
        vector<int> vec;
        while (!q.empty()) {
            vec.push_back(q.front());
            q.pop();
        }
        int left = 0, right = vec.size() - 1;
        while (left < right) {
            if (vec[left] != vec[right]) return false;
            left++; right--;
        }
        return true;
    };
    
    cout << "Очередь 1 палиндром? " << (isPalindrome(q1) ? "Да" : "Нет") << endl;
    cout << "Очередь 2 палиндром? " << (isPalindrome(q2) ? "Да" : "Нет") << "\n\n";
}

// ==========================================
// 3. ДЕК: Удалить все нечётные элементы
// ==========================================
void taskDeque() {
    cout << "--- 3. ДЕК (удаление нечетных) ---" << endl;
    deque<int> dq = {10, 21, 32, 43, 54, 65}; // Исходный дек
    
    cout << "Исходный дек: ";
    for (int x : dq) cout << x << " ";
    cout << endl;
    
    // Итератор для прохода по деку
    for (auto it = dq.begin(); it != dq.end(); ) {
        if (*it % 2 != 0) {
            it = dq.erase(it); // erase возвращает итератор на следующий элемент
        } else {
            ++it;
        }
    }
    
    cout << "Дек после удаления нечетных: ";
    for (int x : dq) cout << x << " ";
    cout << "\n\n";
}

void taskPriorityQueue() {
    cout << "--- 4. ПРИОРИТЕТНАЯ ОЧЕРЕДЬ (min-heap) ---" << endl;
    
    // По умолчанию priority_queue в C++ - это max-heap (самый большой элемент сверху).
    // Чтобы сделать min-heap (самый маленький сверху), используются компараторы:
    
    // Вариант 1 (современный): greater<int>
    priority_queue<int, vector<int>, greater<int>> minHeap;
    
    minHeap.push(5);
    minHeap.push(2);
    minHeap.push(8);
    minHeap.push(1);
    minHeap.push(9);
    
    cout << "Извлечение из min-heap (должно идти по возрастанию): ";
    while (!minHeap.empty()) {
        cout << minHeap.top() << " ";
        minHeap.pop();
    }

}

**Контрольные вопросы**

Принцип LIFO (Last In, First Out) означает «последним пришёл — первым вышел». Элемент, добавленный в стек последним, будет извлечён из него первым. Добавление и извлечение происходят с одного конца (вершины стека).

Принцип FIFO (First In, First Out) означает «первым пришёл — первым вышел». Элемент, добавленный в очередь первым, будет извлечён из неё первым. Добавление идёт в конец очереди, а извлечение — из начала.

Дек (двусторонняя очередь) позволяет добавлять и удалять элементы с обоих концов (как с начала, так и с конца). В обычной очереди добавление возможно только в конец, а удаление — только из начала.

Приоритетная очередь — это структура данных, где элементы извлекаются не в порядке добавления, а в соответствии с их приоритетом (сначала извлекается элемент с наивысшим приоритетом). Применяется в алгоритмах на графах (Дейкстра, Прима), при планировании задач в ОС и в симуляциях.

Стек на динамическом массиве реализуется хранением массива и переменной-указателя на вершину (индекса последнего элемента). При добавлении (push) индекс увеличивается, при извлечении (pop) — уменьшается. При переполнении массива выделяется новый блок памяти большего размера.

Очередь на кольцевом буфере реализуется с помощью массива фиксированного размера и двух индексов: head (начало) и tail (конец). При добавлении элемента индекс tail сдвигается вперёд, при извлечении — head. Когда индексы доходят до конца массива, они «заворачиваются» в начало (используется деление по модулю).

Приоритетную очередь обычно реализуют на куче (обычно двоичной), потому что куча обеспечивает эффективное выполнение основных операций: вставка и извлечение максимума (или минимума) выполняются за O(log n), что значительно быстрее, чем в обычном массиве или связном списке.

В стеке сложность обеих операций push и pop составляет O(1) (амортизированно для динамического массива). Добавление и удаление происходят строго с одного конца, не требуя сдвига остальных элементов.

В очереди сложность операций enqueue (добавление в конец) и dequeue (извлечение из начала) составляет O(1). Это достигается за счёт использования указателей на начало и конец или кольцевого буфера.

В приоритетной очереди (на куче) сложность операций push (вставка) и pop (извлечение) составляет O(log n). Это связано с тем, что после добавления или удаления элемента необходимо восстановить свойство кучи (просеивание вверх или вниз), глубина которой равна логарифму от количества элементов.

Для проверки скобок стек используется так: при встрече открывающей скобки она добавляется в стек; при встрече закрывающей скобки проверяется, лежит ли на вершине стека соответствующая открывающая скобка. Если да — она удаляется, если нет — расстановка неверна. В конце стек должен быть пуст.

Очередь на двух стеках реализуется так: один стек (inbox) используется для добавления элементов (enqueue), второй (outbox) — для извлечения (dequeue). Если outbox пуст при извлечении, в него перекладываются все элементы из inbox (при этом порядок меняется на обратный). Амортизированная сложность обеих операций O(1), хотя перекладывание занимает O(n).

Дек на основе двусвязного списка реализуется хранением указателей на первый и последний узлы. Каждый узел хранит указатели на следующий и предыдущий элементы. Добавление и удаление с обоих концов выполняются за O(1) путём изменения ссылок у головы или хвоста.

Дек на кольцевом буфере реализуется как массив с двумя индексами (head и tail), которые могут двигаться в обоих направлениях. В отличие от обычной очереди, добавление и удаление могут происходить с любой стороны, при этом индексы заворачиваются по модулю размера массива.

Стек с поддержкой минимума за O(1) реализуется с помощью двух стеков: в основной стек добавляются все элементы, а во вспомогательный (стек минимумов) добавляется либо текущий минимум, либо новый элемент, если он меньше текущего минимума. При извлечении элемента из основного стека, он также извлекается из вспомогательного.
