```cpp
using namespace std;

// 1. ФАКТОРИАЛ ЧИСЛА
// Рекурсивная функция вычисления факториала n! = n * (n-1)!
unsigned long long factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

// 2. ЧИСЛО СОЧЕТАНИЙ C(n, k)
// Формула: C(n, k) = n! / (k! * (n-k)!)
unsigned long long combinations(int n, int k) {
    if (k < 0 || k > n) return 0;
    if (k == 0 || k == n) return 1;
    if (k > n / 2) k = n - k;
    
    unsigned long long res = 1;
    for (int i = 1; i <= k; ++i) {
        res = res * (n - i + 1) / i;
    }
    return res;
}

// 3. РАЗЛОЖЕНИЕ НА СЛАГАЕМЫЕ
// Вывод всех способов представить число n в виде суммы слагаемых (невозрастающих)
void printPartitions(int n, int max_val, vector<int>& current) {
    if (n == 0) {
        cout << current[0];
        for (size_t i = 1; i < current.size(); ++i) {
            cout << " + " << current[i];
        }
        cout << endl;
        return;
    }
    
    int start = min(n, max_val);
    for (int i = start; i >= 1; --i) {
        current.push_back(i);
        printPartitions(n - i, i, current);
        current.pop_back();
    }
}

// 4. ГЕНЕРАЦИЯ ВСЕХ ПОДМНОЖЕСТВ
void generateSubsets(const vector<int>& arr, int index, vector<int>& current) {
    if (index == arr.size()) {
        cout << "{ ";
        for (int x : current) cout << x << " ";
        cout << "}" << endl;
        return;
    }
    
    generateSubsets(arr, index + 1, current);
    
    current.push_back(arr[index]);
    generateSubsets(arr, index + 1, current);
    current.pop_back();
}
```

```cpp
#include <iostream>
#include <vector>

using namespace std;

// 1. ПУЗЫРЬКОВАЯ СОРТИРОВКА (Bubble Sort)
void bubbleSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; ++i) {
        bool swapped = false;
        for (int j = 0; j < n - i - 1; ++j) {
            if (arr[j] > arr[j + 1]) {
                swap(arr[j], arr[j + 1]);
                swapped = true;
            }
        }
        if (!swapped) break;
    }
}

// 2. СОРТИРОВКА ВСТАВКАМИ (Insertion Sort)
void insertionSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 1; i < n; ++i) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
    }
}

// 3. ОБЫЧНАЯ СОРТИРОВКА ВЫБОРОМ (Selection Sort)
void selectionSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; ++i) {
        int minIndex = i;
        for (int j = i + 1; j < n; ++j) {
            if (arr[j] < arr[minIndex]) {
                minIndex = j;
            }
        }
        if (minIndex != i) {
            swap(arr[i], arr[minIndex]);
        }
    }
}

// 4. УСТОЙЧИВАЯ СОРТИРОВКА ВЫБОРОМ (Stable Selection Sort)
void stableSelectionSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; ++i) {
        int minIndex = i;
        for (int j = i + 1; j < n; ++j) {
            if (arr[j] < arr[minIndex]) {
                minIndex = j;
            }
        }
        int key = arr[minIndex];
        while (minIndex > i) {
            arr[minIndex] = arr[minIndex - 1];
            minIndex--;
        }
        arr[i] = key;
    }
}

void printArray(const vector<int>& arr) {
    for (int x : arr) {
        cout << x << " ";
    }
    cout << endl;
}

int main() {
    setlocale(LC_ALL, "ru");

    vector<int> original = {64, 25, 12, 22, 11, 12, 90, 5};

    cout << "Исходный массив: ";
    printArray(original);

    vector<int> arr1 = original;
    bubbleSort(arr1);
    cout << "\n1. Пузырьковая сортировка: ";
    printArray(arr1);

    vector<int> arr2 = original;
    insertionSort(arr2);
    cout << "2. Сортировка вставками:   ";
    printArray(arr2);

    vector<int> arr3 = original;
    selectionSort(arr3);
    cout << "3. Обычная сортировка выбором: ";
    printArray(arr3);

    vector<int> arr4 = original;
    stableSelectionSort(arr4);
    cout << "4. Устойчивая сортировка выбором: ";
    printArray(arr4);

    return 0;
}
```

**Контрольные вопросы**

Рекурсия — это метод решения задачи, при котором функция вызывает саму себя. Рекурсивная функция состоит из базового случая (условия остановки) и рекурсивного шага (вызова самой себя с измененными параметрами, приближающими к базовому случаю).

Базовый случай — это условие, при котором функция прекращает вызывать себя и возвращает конкретное значение. Он важен, потому что без него рекурсия станет бесконечной и приведет к переполнению стека (Stack Overflow).

При прямой рекурсии функция вызывает саму себя напрямую (например, функция А вызывает функцию А). При косвенной рекурсии функция вызывает другую функцию, которая в итоге снова вызывает первую (например, А вызывает В, а В вызывает А).

Хвостовая рекурсия — это рекурсивный вызов, который является последней операцией в функции (после него нет никаких вычислений). Её преимущество в том, что умные компиляторы могут оптимизировать её до обычного цикла, экономя память на стеке вызовов.

Временная сложность наивного рекурсивного вычисления чисел Фибоначчи — О(2 в степени n) (экспоненциальная). Это происходит потому, что каждый вызов функции порождает два новых вызова, создавая огромное дерево повторных вычислений одних и тех же значений.

Оптимизировать вычисление Фибоначчи можно с помощью мемоизации (сохранения уже вычисленных результатов в массиве или хеш-таблице) или переписав алгоритм итеративно. В обоих случаях сложность снижается до O(n).

Рекурсивная реализация обычно более короткая, наглядная и ближе к математическому описанию задачи, но требует дополнительной памяти под стек вызовов. Итеративная реализация использует циклы, работает быстрее (нет накладных расходов на вызовы) и не требует лишней памяти.

Сложность пузырьковой сортировки в худшем и среднем случае — О(n в квадрате). Она становится О(n) в лучшем случае (на уже отсортированном массиве), если добавить флаг проверки, были ли обмены за текущий проход. Если обменов не было, алгоритм завершается досрочно.

Сортировка вставками вставляет каждый элемент в уже отсортированную часть, сдвигая элементы. Сортировка выбором на каждом шаге ищет минимальный элемент в неотсортированной части и меняет его местами с текущим. Вставки лучше работают на почти отсортированных данных.

Из простых сортировок устойчивыми являются пузырьковая и вставками. Устойчивость означает, что элементы с одинаковыми значениями сохраняют свой исходный относительный порядок после сортировки.

Лучше всего на почти отсортированных данных работает сортировка вставками. Если массив почти отсортирован, внутреннему циклу почти не приходится сдвигать элементы, и алгоритм выполняется близко к О(n).

Для реализации оптимизированной пузырьковой сортировки нужно ввести логическую переменную-флаг (например, swapped). В начале каждого прохода она равна false. Если происходит обмен, флаг становится true. Если после прохода флаг остался false, значит массив уже отсортирован, и цикл прерывается.

Рекурсивный поиск с возвратом (backtracking) — это метод решения задач, при котором мы строим решение шаг за шагом. Если на каком-то этапе становится ясно, что решение неверно, мы отменяем последний шаг (возвращаемся назад) и пробуем другой вариант.

Рекурсивный обход лабиринта: начиная с текущей клетки, мы пробуем пойти во все возможные стороны (вверх, вниз, влево, вправо). Если соседняя клетка проходима и мы в ней еще не были, мы рекурсивно вызываем функцию для неё. Если дошли до выхода — возвращаем успех, иначе откатываемся назад.

Сложность Ханойских башен — O(2 в степени n) (экспоненциальная). Для перемещения n дисков с одного стержня на другой требуется ровно 2 в степени n минус 1 ходов. Это минимально возможное количество ходов для данной задачи.
