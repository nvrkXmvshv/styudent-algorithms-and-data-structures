#include <iostream>
#include <vector>
#include <chrono>
#include <algorithm>
#include <random>

using namespace std;
using namespace std::chrono;

// ==========================================
// СОРТИРОВКА СЛИЯНИЕМ (Merge Sort)
// ==========================================

// Функция слияния двух отсортированных половин
void merge(vector<int>& arr, int left, int mid, int right) {
    int n1 = mid - left + 1;
    int n2 = right - mid;

    // Создаем временные массивы
    vector<int> L(n1), R(n2);

    for (int i = 0; i < n1; i++) L[i] = arr[left + i];
    for (int j = 0; j < n2; j++) R[j] = arr[mid + 1 + j];

    int i = 0, j = 0, k = left;

    // Сливаем обратно в arr
    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) {
            arr[k] = L[i];
            i++;
        } else {
            arr[k] = R[j];
            j++;
        }
        k++;
    }

    // Копируем оставшиеся элементы
    while (i < n1) {
        arr[k] = L[i];
        i++;
        k++;
    }
    while (j < n2) {
        arr[k] = R[j];
        j++;
        k++;
    }
}

// Основная рекурсивная функция
void mergeSort(vector<int>& arr, int left, int right) {
    if (left >= right) return;

    int mid = left + (right - left) / 2;
    
    mergeSort(arr, left, mid);
    mergeSort(arr, mid + 1, right);

    // Оптимизация для почти отсортированных массивов:
    // Если последний элемент левой части меньше или равен первому элементу правой,
    // значит, массив уже отсортирован, сливать не нужно.
    if (arr[mid] <= arr[mid + 1]) return;

    merge(arr, left, mid, right);
}

vector<int> generateNearlySorted(int size) {
    vector<int> arr(size);
    for (int i = 0; i < size; i++) arr[i] = i; // Уже отсортирован

    random_device rd;
    mt19937 g(rd());

    int swaps = size * 0.05; 
    for (int i = 0; i < swaps; i++) {
        int idx = rand() % (size - 1);
        swap(arr[idx], arr[idx + 1]);
    }
    return arr;
}

// Замер времени
template <typename Func>
long long measureTime(Func func, vector<int>& arr) {
    auto start = high_resolution_clock::now();
    func(arr, 0, arr.size() - 1);
    auto end = high_resolution_clock::now();
    return duration_cast<microseconds>(end - start).count();
}

Контрольные вопросы:

1. В лучшем и среднем случае сложность быстрой сортировки — O(n log n). В худшем случае — O(n в квадрате).
2. Худший случай возникает, когда опорный элемент (pivot) на каждом шаге выбирается крайне неудачно (например, всегда минимальный или максимальный элемент). Это приводит к тому, что массив делится на две части: пустую и размером n-1. Такое часто случается, если массив уже отсортирован, а опорным элементом всегда выбирается первый или последний.
3. Для защиты от худшего случая используют случайный выбор опорного элемента (randomized quicksort), медиану из трех (выбор медианы среди первого, среднего и последнего элементов) или интроспективную сортировку (переключение на другую сортировку при глубокой рекурсии).
4. Сложность сортировки слиянием — O(n log n) во всех случаях. Это происходит потому, что массив всегда делится строго пополам (глубина рекурсии всегда равна log n), а на каждом уровне рекурсии выполняется операция слияния, которая суммарно занимает O(n) времени.
5. Преимущества сортировки слиянием: предсказуемое время O(n log n) в любом случае, устойчивость. Недостатки: требует O(n) дополнительной памяти для временных массивов, на практике часто работает медленнее быстрой сортировки из-за накладных расходов на копирование данных.
6. Пирамидальная сортировка (Heap sort) работает так: сначала из массива строится бинарная куча (max-heap), затем самый большой элемент (корень) меняется местами с последним элементом массива, а размер кучи уменьшается. Затем куча восстанавливается (просеивание вниз). Сложность — O(n log n) во всех случаях.
7. Из этих трёх сортировок устойчивой является только сортировка слиянием. Она сохраняет относительный порядок равных элементов, так как при слиянии мы всегда выбираем элемент из левой части, если элементы равны. Быстрая и пирамидальная сортировки обменивают элементы на большие расстояния, нарушая исходный порядок.
8. Дополнительной памяти требует сортировка слиянием — O(n) (для временных массивов при слиянии). Быстрая сортировка требует O(log n) памяти на стек рекурсии. Пирамидальная сортировка работает in-place и требует O(1) памяти.
9. Лучше всего на почти отсортированных данных работает быстрая сортировка (если выбран хороший опорный элемент, например, медиана). Она будет выполнять меньше обменов. Сортировка слиянием будет работать так же, как и на случайных данных (из-за проверки `if (arr[mid] <= arr[mid + 1])` она ускорится, но всё равно будет O(n log n)).
10. Куча строится для того, чтобы эффективно извлекать максимальный (или минимальный) элемент. Построенная куча гарантирует, что корень всегда содержит максимум, и его извлечение занимает O(log n), а не O(n) как при линейном поиске.
11. Метод «разделяй и властвуй» заключается в разбиении задачи на несколько более мелких подзадач, их решении и объединении результатов. В быстрой сортировке мы делим массив на две части относительно опорного элемента. В сортировке слиянием мы делим массив пополам, сортируем каждую часть, а затем сливаем.
12. Да, быструю сортировку можно реализовать итеративно, используя собственный стек (например, `std::stack`) для хранения границ подмассивов, которые еще нужно отсортировать. Это позволяет избежать переполнения системного стека при большой глубине рекурсии.
13. Быстрая сортировка обычно быстрее на практике из-за меньших накладных расходов: она работает "на месте" (не выделяет память под новые массивы), имеет лучшую локальность кэша (работает с последовательными участками памяти) и делает меньше операций копирования.
