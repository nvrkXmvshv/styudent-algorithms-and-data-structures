```cpp
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

// Структура Person
struct Person {
    string name;
    int age;
};

// Функция для вывода массива людей
void printPeople(const vector<Person>& people) {
    for (const auto& p : people) {
        cout << p.name << " (" << p.age << " лет)" << endl;
    }
}

// Функция двоичного поиска по возрасту
// Возвращает индекс найденного человека или -1
int binarySearchByAge(const vector<Person>& people, int targetAge) {
    int left = 0;
    int right = people.size() - 1;

    while (left <= right) {
        int mid = left + (right - left) / 2;

        if (people[mid].age == targetAge) {
            return mid; // Нашли!
        }
        if (people[mid].age < targetAge) {
            left = mid + 1; // Ищем в правой половине
        } else {
            right = mid - 1; // Ищем в левой половине
        }
    }
    return -1; // Не нашли
}

int main() {
    setlocale(LC_ALL, "ru");

    // Исходный массив (не отсортирован по возрасту)
    vector<Person> people = {
        {"Иван", 25},
        {"Мария", 19},
        {"Петр", 45},
        {"Анна", 30},
        {"Сергей", 22},
        {"Елена", 35}
    };

    cout << "--- Исходный список ---" << endl;
    printPeople(people);

    // ШАГ 1: Сортировка по возрасту (обязательно для бинарного поиска!)
    sort(people.begin(), people.end(), [](const Person& a, const Person& b) {
        return a.age < b.age;
    });

    cout << "\n--- Список после сортировки по возрасту ---" << endl;
    printPeople(people);

    // ШАГ 2: Поиск по ключу (возрасту)
    int searchAge = 30;
    cout << "\n--- Поиск человека с возрастом " << searchAge << " ---" << endl;
    
    int index = binarySearchByAge(people, searchAge);
    if (index != -1) {
        cout << "Найден: " << people[index].name << ", возраст: " << people[index].age << endl;
    } else {
        cout << "Человек с возрастом " << searchAge << " не найден." << endl;
    }

    // Проверка поиска несуществующего элемента
    searchAge = 99;
    cout << "\n--- Поиск человека с возрастом " << searchAge << " ---" << endl;
    index = binarySearchByAge(people, searchAge);
    if (index != -1) {
        cout << "Найден: " << people[index].name << ", возраст: " << people[index].age << endl;
    } else {
        cout << "Человек с возрастом " << searchAge << " не найден." << endl;
    }

    return 0;
}
```

Вариант	Метод разрешения коллизий	Тип ключа	Тип значения

```cpp
#include <iostream>
#include <string>
#include <vector>

using namespace std;

class HashTable {
private:
    struct Entry {
        int key;
        string value;
        bool isDeleted;

        Entry() : key(0), value(""), isDeleted(false) {}
    };

    vector<Entry> table;
    int size;       // Текущее количество активных элементов
    int capacity;   // Общий размер массива

    // Хеш-функция (метод деления)
    int hashFunc(int key) {
        return key % capacity;
    }

    // Рехеширование (расширение таблицы)
    void rehash() {
        vector<Entry> oldTable = table;
        capacity *= 2; // Увеличиваем размер в 2 раза
        table.clear();
        table.resize(capacity);
        size = 0;

        // Пересыпаем старые элементы в новую таблицу
        for (const auto& entry : oldTable) {
            if (entry.value != "" && !entry.isDeleted) {
                insert(entry.key, entry.value);
            }
        }
    }

public:
    HashTable(int initialCapacity = 10) {
        capacity = initialCapacity;
        size = 0;
        table.resize(capacity);
    }

    // Вставка элемента
    void insert(int key, const string& value) {
        // Если таблица заполнена более чем на 70%, расширяем её
        if (size >= capacity * 0.7) {
            rehash();
        }

        int index = hashFunc(key);
        int startIndex = index;

        // Ищем свободную ячейку или ячейку, помеченную как удаленная
        while (table[index].value != "" && !table[index].isDeleted) {
            if (table[index].key == key) {
                // Если ключ уже есть, обновляем значение
                table[index].value = value;
                return;
            }
            index = (index + 1) % capacity; // Линейное пробирование
            
            // Защита от бесконечного цикла, если таблица полна
            if (index == startIndex) {
                cout << "Ошибка: таблица переполнена!" << endl;
                return;
            }
        }

        // Вставляем новый элемент
        table[index].key = key;
        table[index].value = value;
        table[index].isDeleted = false;
        size++;
    }

    // Поиск элемента по ключу
    string find(int key) {
        int index = hashFunc(key);
        int startIndex = index;

        while (table[index].value != "") {
            if (table[index].key == key && !table[index].isDeleted) {
                return table[index].value; // Нашли
            }
            index = (index + 1) % capacity;
            
            if (index == startIndex) break; // Прошли круг
        }
        return ""; // Не нашли
    }

    // Удаление элемента (ленивое удаление)
    void remove(int key) {
        int index = hashFunc(key);
        int startIndex = index;

        while (table[index].value != "") {
            if (table[index].key == key && !table[index].isDeleted) {
                table[index].isDeleted = true; // Помечаем как удаленное
                size--;
                return;
            }
            index = (index + 1) % capacity;
            
            if (index == startIndex) break;
        }
        cout << "Ключ " << key << " не найден для удаления." << endl;
    }

    // Вывод содержимого таблицы
    void print() {
        cout << "--- Хеш-таблица (Размер: " << size << ", Емкость: " << capacity << ") ---" << endl;
        for (int i = 0; i < capacity; i++) {
            cout << i << ": ";
            if (table[i].value == "") {
                cout << "[пусто]" << endl;
            } else if (table[i].isDeleted) {
                cout << "[удалено]" << endl;
            } else {
                cout << "{" << table[i].key << " : " << table[i].value << "}" << endl;
            }
        }
        cout << endl;
    }
};

int main() {
    setlocale(LC_ALL, "ru");

    HashTable ht(10);

    cout << "--- Вставка элементов ---" << endl;
    ht.insert(10, "Десять");
    ht.insert(20, "Двадцать");
    ht.insert(15, "Пятнадцать"); // Коллизия с 20? (20 % 10 = 0, 15 % 10 = 5)
    ht.insert(25, "Двадцать пять"); // Коллизия с 15 (25 % 10 = 5)
    ht.insert(12, "Двенадцать");
    ht.print();

    cout << "--- Поиск ---" << endl;
    cout << "Поиск ключа 15: " << ht.find(15) << endl;
    cout << "Поиск ключа 25: " << ht.find(25) << endl;
    cout << "Поиск ключа 99: " << (ht.find(99) == "" ? "не найден" : ht.find(99)) << endl;
    cout << endl;

    cout << "--- Удаление ключа 20 ---" << endl;
    ht.remove(20);
    ht.print();

    cout << "--- Проверка после удаления (ленивое удаление не рвет цепочку) ---" << endl;
    cout << "Поиск ключа 10 (был до удаленного): " << ht.find(10) << endl;
    cout << "Поиск ключа 15 (после удаленного): " << ht.find(15) << endl;

    return 0;
}
```

**Контрольные вопросы**

Для применения двоичного поиска необходимо, чтобы массив данных был предварительно отсортирован (например, по возрастанию). Кроме того, должен быть обеспечен быстрый доступ к элементам по индексу (O(1)).

Сложность двоичного поиска — O(log n). Она логарифмическая, потому что на каждом шаге алгоритм отбрасывает половину оставшихся элементов. Количество шагов равно тому, сколько раз нужно разделить n пополам, чтобы дойти до 1, что и есть логарифм по основанию 2.

Итеративная реализация использует цикл (while или for) и не требует дополнительной памяти (работает за O(1) памяти). Рекурсивная реализация вызывает саму себя, что требует O(log n) памяти под стек вызовов, но выглядит более компактно и наглядно.

Вариант mid = (left + right) / 2 опасен тем, что при очень больших значениях left и right может произойти целочисленное переполнение, что приведет к отрицательному индексу и ошибке. Правильно вычислять середину так: mid = left + (right - left) / 2.

Хеш-таблица — это структура данных для хранения пар «ключ-значение». Она состоит из массива (или списков) и хеш-функции, которая преобразует ключ в индекс массива.

Хеш-функция должна быть детерминированной (один и тот же ключ всегда дает один и тот же хеш), быстрой (O(1)) и равномерно распределять ключи по всему массиву, чтобы минимизировать коллизии.

Коллизия — это ситуация, когда два разных ключа в результате хеш-функции получают одинаковый индекс. С ней борются двумя основными способами: методом цепочек (каждая ячейка хранит список элементов) или открытой адресацией (поиск следующей свободной ячейки).

Метод цепочек хранит коллизионные элементы в виде связного списка в одной ячейке. Открытая адресация хранит элементы непосредственно в самом массиве и при занятости ячейки ищет следующую свободную по определенному правилу (линейное пробирование, квадратичное и т.д.).

Первичная кластеризация — это образование длинных непрерывных цепочек занятых ячеек при линейном пробировании, что сильно замедляет поиск. Вторичная кластеризация возникает при квадратичном пробировании, когда цепочки хоть и не сливаются в одну длинную, но образуют локальные группы.

Средняя сложность операций поиска, вставки и удаления в хеш-таблице — O(1). Худшая сложность — O(n), когда все элементы попали в одну ячейку (например, при плохой хеш-функции).

Двоичный поиск эффективнее, когда массив уже отсортирован, а данные статичны (не меняются), и требуется экономить память. Хеш-таблица эффективнее, когда требуются частые вставки, удаления и поиск за константное время, и можно выделить дополнительную память.

Коэффициент загрузки — это отношение количества элементов в таблице к общему размеру массива (n/m). Он важен, потому что при его высоком значении (близком к 1) резко возрастает количество коллизий и время поиска. Обычно при достижении 0.7-0.75 таблицу расширяют.

Рехеширование — это процесс создания нового, большего по размеру массива и пересчета хеш-функций для всех существующих элементов с последующим переносом их в новую таблицу. Это дорогая операция (O(n)), но она выполняется редко.

Размер хеш-таблицы обычно выбирают так, чтобы он был в 1.5-2 раза больше ожидаемого количества элементов. Часто выбирают простое число, чтобы минимизировать коллизии при использовании метода деления (так как простое число не имеет общих делителей с большинством ключей).

Для строк хеш-функция вычисляется путем перемножения кодов символов на степени некоторого простого числа (например, 31 или 37) с последующим сложением. Например, хеш = (s[0]p^0 + s[1]p^1 + ... + s[k]*p^k) mod m.
